3) Ligand-protein contact occupancy

Install MDAnalysis:

conda create -n mda python=3.11 -c conda-forge -y

Script (Ligand_contacts.py):

import MDAnalysis as mda
from MDAnalysis.analysis import distances
import numpy as np
from collections import defaultdict

u = mda.Universe("step7_production.gro", "step7_production_fit.xtc")

protein = u.select_atoms("protein")
ligand = u.select_atoms("resname UNL")

cutoff = 4.0  # Angstrom

contact_counts = defaultdict(int)
nframes = 0

for ts in u.trajectory:
    nframes += 1
    for res in protein.residues:
        d = distances.distance_array(
            res.atoms.positions, ligand.positions, box=u.dimensions
        )
        if np.any(d < cutoff):
            contact_counts[f"{res.resname}{res.resid}"] += 1

print("Residue\tOccupancy(%)")
for res, count in sorted(contact_counts.items(), key=lambda x: x[1], reverse=True):
    occ = 100 * count / nframes
    if occ >= 1:
        print(f"{res}\t{occ:.2f}")

python Ligand_contacts.py > contact_occupancy.txt
